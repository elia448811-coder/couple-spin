rules_version = '2';
// DOUBLE_GAME_RULES_v32_SECURITY_ACTIVE_ACCOUNT
service cloud.firestore {
  match /databases/{database}/documents {
    function isAuth() {
      return request.auth != null;
    }

    // Sole admin: locked UID and/or locked email from Auth token (not spoofable).
    function isAdmin() {
      return isAuth() && (
        request.auth.uid == 'tpbKWXtXWFapFC7Fd80Wd4IMqxC2'
        || (
          request.auth.token.email is string
          && request.auth.token.email.lower() == 'elia448811@gmail.com'
        )
      );
    }

    // Real account (blocks anonymous Auth used only for bootstrap).
    function isEmailSignedIn() {
      return isAuth()
        && request.auth.token.email is string
        && request.auth.token.email.size() > 3
        && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    function callerUserDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    // Enforced server-side: profile must exist, not banned, not explicitly unapproved.
    // Legacy docs without `approved` remain allowed (default true).
    // Missing profile is NOT active — prevents Auth-only bypass of approval.
    function isActiveAccount() {
      return isAdmin() || (
        isEmailSignedIn()
        && callerUserDoc().exists
        && callerUserDoc().data.get('banned', false) == false
        && callerUserDoc().data.get('approved', true) != false
      );
    }

    function roomDoc(roomId) {
      return get(/databases/$(database)/documents/rooms/$(roomId));
    }

    function isRoomMember(roomId) {
      return isAuth()
        && roomDoc(roomId).exists
        && (
          roomDoc(roomId).data.hostUid == request.auth.uid
          || roomDoc(roomId).data.partnerUid == request.auth.uid
        );
    }

    function isRoomHost(roomId) {
      return isAuth()
        && roomDoc(roomId).exists
        && roomDoc(roomId).data.hostUid == request.auth.uid;
    }

    function roomResourceActive() {
      return resource.data.expiresAtMs > request.time.toMillis();
    }

    function roomActive(roomId) {
      return roomDoc(roomId).exists
        && roomDoc(roomId).data.expiresAtMs > request.time.toMillis();
    }

    function unchanged(field) {
      return !(field in request.resource.data)
        || request.resource.data[field] == resource.data[field];
    }

    function validDisplayCode(code) {
      return code.matches('^[0-9]{8}$');
    }

    function lockedRoomIdentity() {
      return unchanged('hostUid')
        && unchanged('displayCode')
        && unchanged('createdAtMs')
        && unchanged('expiresAtMs');
    }

    function validRoomStatus(value) {
      return value in ['waiting', 'lobby', 'playing', 'ended'];
    }

    function hostPresenceUpdate() {
      return resource.data.hostUid == request.auth.uid
        && unchanged('partnerUid')
        && unchanged('hostName')
        && unchanged('partnerName')
        && unchanged('eveningTitle')
        && unchanged('version')
        && unchanged('status')
        && unchanged('lastEventId')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['hostReady', 'hostLastSeenMs', 'updatedAtMs', 'updatedBy', 'updatedAt']);
    }

    function partnerPresenceUpdate() {
      return resource.data.partnerUid == request.auth.uid
        && unchanged('partnerUid')
        && unchanged('hostName')
        && unchanged('partnerName')
        && unchanged('eveningTitle')
        && unchanged('version')
        && unchanged('status')
        && unchanged('lastEventId')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['partnerReady', 'partnerLastSeenMs', 'updatedAtMs', 'updatedBy', 'updatedAt']);
    }

    function hostVersionBump() {
      return resource.data.hostUid == request.auth.uid
        && request.resource.data.version == resource.data.version + 1
        && unchanged('partnerUid')
        && unchanged('hostName')
        && unchanged('partnerName')
        && unchanged('eveningTitle')
        && unchanged('hostReady')
        && unchanged('partnerReady')
        && unchanged('hostLastSeenMs')
        && unchanged('partnerLastSeenMs')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['version', 'lastEventId', 'updatedBy', 'updatedAtMs', 'updatedAt', 'status'])
        && validRoomStatus(request.resource.data.status);
    }

    function partnerVersionBump() {
      return resource.data.partnerUid == request.auth.uid
        && request.resource.data.version == resource.data.version + 1
        && unchanged('partnerUid')
        && unchanged('hostName')
        && unchanged('partnerName')
        && unchanged('eveningTitle')
        && unchanged('hostReady')
        && unchanged('partnerReady')
        && unchanged('hostLastSeenMs')
        && unchanged('partnerLastSeenMs')
        && unchanged('status')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['version', 'lastEventId', 'updatedBy', 'updatedAtMs', 'updatedAt']);
    }

    function partnerJoinUpdate() {
      return resource.data.partnerUid == null
        && resource.data.status == 'waiting'
        && request.auth.uid != resource.data.hostUid
        && request.resource.data.partnerUid == request.auth.uid
        && request.resource.data.partnerName is string
        && request.resource.data.partnerName.size() > 0
        && request.resource.data.partnerName.size() <= 32
        && request.resource.data.partnerReady == false
        && request.resource.data.status == 'lobby'
        && unchanged('hostName')
        && unchanged('eveningTitle')
        && unchanged('version')
        && unchanged('lastEventId')
        && unchanged('hostReady')
        && unchanged('hostLastSeenMs')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            'partnerUid',
            'partnerName',
            'partnerReady',
            'partnerLastSeenMs',
            'status',
            'updatedAtMs',
            'updatedBy',
            'updatedAt'
          ]);
    }

    function partnerLeaveUpdate() {
      return resource.data.partnerUid == request.auth.uid
        && request.resource.data.partnerUid == null
        && request.resource.data.partnerName == null
        && request.resource.data.partnerReady == false
        && request.resource.data.partnerLastSeenMs == 0
        && request.resource.data.status == 'waiting'
        && unchanged('hostName')
        && unchanged('eveningTitle')
        && unchanged('version')
        && unchanged('lastEventId')
        && unchanged('hostReady')
        && unchanged('hostLastSeenMs')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            'partnerUid',
            'partnerName',
            'partnerReady',
            'partnerLastSeenMs',
            'status',
            'updatedAtMs',
            'updatedBy',
            'updatedAt'
          ]);
    }

    function validEventType() {
      return request.resource.data.type in [
        'PLAYER_READY',
        'GAME_STARTED',
        'GAME_SNAPSHOT',
        'GAME_ENDED',
        'TURN_CHANGED'
      ];
    }

    function partnerMayCreateEvent() {
      return request.resource.data.type == 'PLAYER_READY';
    }

    function gameStartAllowed(roomId) {
      return request.resource.data.type != 'GAME_STARTED'
        || (
          roomDoc(roomId).data.hostReady == true
          && roomDoc(roomId).data.partnerReady == true
          && roomDoc(roomId).data.partnerUid != null
        );
    }

    function rateLimitCreateOk() {
      return request.resource.data.keys().hasOnly(['failedJoins', 'lockedUntilMs', 'updatedAtMs', 'updatedAt'])
        && request.resource.data.failedJoins == 1
        && request.resource.data.lockedUntilMs == 0;
    }

    function rateLimitUpdateOk() {
      return request.resource.data.failedJoins is int
        && request.resource.data.lockedUntilMs is int
        && (
          (
            request.resource.data.failedJoins == resource.data.get('failedJoins', 0) + 1
            && request.resource.data.lockedUntilMs == resource.data.get('lockedUntilMs', 0)
            && request.resource.data.failedJoins <= 20
          )
          || (
            request.resource.data.failedJoins == 0
            && request.resource.data.lockedUntilMs > resource.data.get('lockedUntilMs', 0)
            && request.resource.data.lockedUntilMs >= request.time.toMillis()
            && resource.data.get('failedJoins', 0) >= 9
          )
          || (
            resource.data.get('lockedUntilMs', 0) > 0
            && resource.data.get('lockedUntilMs', 0) <= request.time.toMillis()
            && request.resource.data.failedJoins == 0
            && request.resource.data.lockedUntilMs == 0
          )
        );
    }

    function validUserProfile() {
      return request.resource.data.uid == request.auth.uid
        && request.resource.data.displayName is string
        && request.resource.data.displayName.size() > 0
        && request.resource.data.displayName.size() <= 32
        && request.resource.data.partnerDisplayName is string
        && request.resource.data.partnerDisplayName.size() <= 32
        && request.resource.data.lastSeenMs is int
        && request.resource.data.updatedAtMs is int
        && request.resource.data.schemaVersion is int;
    }

    function userCreateSafe() {
      return (!('banned' in request.resource.data) || request.resource.data.banned == false)
        && (!('approved' in request.resource.data) || request.resource.data.approved == false)
        && (!('adminNote' in request.resource.data) || request.resource.data.adminNote == '');
    }

    function userSelfFieldLocks() {
      return unchanged('banned')
        && unchanged('approved')
        && unchanged('adminNote');
    }

    function userSelfUpdateKeysOk() {
      return request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly([
          'displayName',
          'partnerDisplayName',
          'avatar',
          'lastSeenMs',
          'updatedAtMs',
          'updatedAt',
          'gamesPlayed',
          'lastRoomId',
          'username',
          'schemaVersion',
          'createdAt',
          'createdAtMs'
        ]);
    }

    // Public site settings (login page needs registration toggle before auth)
    match /config/{docId} {
      allow read: if true;
      allow create, update: if isAdmin()
        && request.resource.data.updatedBy == request.auth.uid;
      allow delete: if false;
    }

    // Legacy admins collection — locked forever
    match /admins/{uid} {
      allow read, write: if false;
    }

    // Content bank overrides (questions/tasks edited in admin)
    match /contentItems/{itemId} {
      allow read: if isActiveAccount();
      allow create, update: if isAdmin()
        && request.resource.data.updatedByUid == request.auth.uid;
      allow delete: if isAdmin();
    }

    // Root user account document
    match /users/{userId} {
      allow read: if isEmailSignedIn() && (request.auth.uid == userId || isAdmin());
      allow create: if isEmailSignedIn() && (
        (
          request.auth.uid == userId
          && validUserProfile()
          && userCreateSafe()
        )
        || (
          isAdmin()
          && request.resource.data.uid == userId
          && request.resource.data.displayName is string
          && request.resource.data.displayName.size() > 0
          && request.resource.data.displayName.size() <= 32
        )
      );
      allow update: if isEmailSignedIn()
        && unchanged('uid')
        && unchanged('createdAtMs')
        && (
          (
            request.auth.uid == userId
            && userSelfFieldLocks()
            && userSelfUpdateKeysOk()
            && (
              validUserProfile()
              || request.resource.data.diff(resource.data).affectedKeys()
                .hasOnly(['lastSeenMs', 'updatedAtMs', 'updatedAt', 'gamesPlayed', 'lastRoomId', 'username'])
            )
          )
          || (
            isAdmin()
            // Sole admin account cannot be banned / unapproved via rules.
            && !(userId == 'tpbKWXtXWFapFC7Fd80Wd4IMqxC2'
              && (
                ('banned' in request.resource.data && request.resource.data.banned == true)
                || ('approved' in request.resource.data && request.resource.data.approved == false)
              ))
            && request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly([
                'banned',
                'adminNote',
                'displayName',
                'approved',
                'username',
                'updatedAtMs',
                'updatedAt'
              ])
          )
        );
      allow delete: if false;
    }

    match /users/{userId}/{document=**} {
      allow read: if isEmailSignedIn() && (request.auth.uid == userId || isAdmin());
      allow write: if isEmailSignedIn() && request.auth.uid == userId;
    }

    match /rateLimits/{uid} {
      allow read: if isActiveAccount() && request.auth.uid == uid;
      allow create: if isActiveAccount() && request.auth.uid == uid && rateLimitCreateOk();
      allow update: if isActiveAccount() && request.auth.uid == uid && rateLimitUpdateOk();
      allow delete: if false;
    }

    match /roomCodes/{code} {
      allow read: if isActiveAccount()
        && validDisplayCode(code)
        && resource.data.expiresAtMs > request.time.toMillis();
      allow create: if isActiveAccount()
        && validDisplayCode(code)
        && request.resource.data.roomId is string
        && request.resource.data.hostUid == request.auth.uid
        && request.resource.data.expiresAtMs > request.time.toMillis()
        && request.resource.data.expiresAtMs
          <= request.time.toMillis() + 3 * 60 * 60 * 1000;
      allow delete: if isActiveAccount() && resource.data.hostUid == request.auth.uid;
    }

    match /rooms/{roomId} {
      // Members always; waiting empty rooms readable for join transaction after code lookup.
      allow read: if isActiveAccount()
        && roomResourceActive()
        && (
          resource.data.hostUid == request.auth.uid
          || resource.data.partnerUid == request.auth.uid
          || (
            resource.data.status == 'waiting'
            && resource.data.partnerUid == null
          )
        );

      allow create: if isActiveAccount()
        && request.resource.data.hostUid == request.auth.uid
        && request.resource.data.partnerUid == null
        && request.resource.data.version == 0
        && request.resource.data.expiresAtMs > request.time.toMillis()
        && request.resource.data.expiresAtMs
          <= request.time.toMillis() + 3 * 60 * 60 * 1000
        && request.resource.data.displayCode is string
        && validDisplayCode(request.resource.data.displayCode)
        && request.resource.data.hostName is string
        && request.resource.data.hostName.size() > 0
        && request.resource.data.hostName.size() <= 32
        && request.resource.data.status == 'waiting'
        && request.resource.data.hostReady == false
        && request.resource.data.partnerReady == false;

      allow update: if isActiveAccount() && roomResourceActive() && lockedRoomIdentity() && (
        hostPresenceUpdate()
        || partnerPresenceUpdate()
        || hostVersionBump()
        || partnerVersionBump()
        || partnerJoinUpdate()
        || partnerLeaveUpdate()
      );

      allow delete: if isActiveAccount()
        && resource.data.hostUid == request.auth.uid;

      match /events/{eventId} {
        allow read: if isActiveAccount() && isRoomMember(roomId);
        allow create: if isActiveAccount()
          && isRoomMember(roomId)
          && request.resource.data.createdBy == request.auth.uid;
        allow update, delete: if false;
      }
    }
  }
}
