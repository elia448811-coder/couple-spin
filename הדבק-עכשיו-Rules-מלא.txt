rules_version = '2';
// DOUBLE_GAME_RULES_v33_ACTIVE_ACCOUNT_ROOM_LOOKUP
service cloud.firestore {
  match /databases/{database}/documents {
    function isAuth() {
      return request.auth != null;
    }

    // Sole admin: the pre-provisioned Firebase Auth account.
    function isAdmin() {
      return isAuth() && request.auth.uid == 'tpbKWXtXWFapFC7Fd80Wd4IMqxC2'; // gitleaks:allow public Firebase account identifier
    }

    function registrationOpen() {
      return !exists(/databases/$(database)/documents/config/app)
        || get(/databases/$(database)/documents/config/app).data.get('registrationEnabled', true) == true;
    }

    // Real account (blocks anonymous Auth used only for bootstrap).
    function isEmailSignedIn() {
      return isAuth()
        && request.auth.token.email is string
        && request.auth.token.email.size() > 3
        && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    function callerUserDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    // Account access requires a positive approval recorded by an administrator.
    function isActiveAccount() {
      return isAdmin() || (
        isEmailSignedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && callerUserDoc().data.get('banned', false) == false
        && callerUserDoc().data.get('approved', false) == true
      );
    }

    function roomDoc(roomId) {
      return get(/databases/$(database)/documents/rooms/$(roomId));
    }

    function isRoomMember(roomId) {
      return isAuth()
        && exists(/databases/$(database)/documents/rooms/$(roomId))
        && (
          roomDoc(roomId).data.hostUid == request.auth.uid
          || roomDoc(roomId).data.partnerUid == request.auth.uid
        );
    }

    function roomResourceActive() {
      return resource.data.expiresAtMs > request.time.toMillis();
    }

    function roomActive(roomId) {
      return exists(/databases/$(database)/documents/rooms/$(roomId))
        && roomDoc(roomId).data.expiresAtMs > request.time.toMillis();
    }

    function unchanged(field) {
      return !(field in request.resource.data)
        || request.resource.data[field] == resource.data[field];
    }

    function hostPresenceUpdate() {
      return resource.data.hostUid == request.auth.uid
        && unchanged('partnerUid')
        && unchanged('hostName')
        && unchanged('partnerName')
        && unchanged('eveningTitle')
        && unchanged('version')
        && unchanged('status')
        && unchanged('lastEventId')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['hostReady', 'hostLastSeenMs', 'updatedAtMs', 'updatedBy', 'updatedAt']);
    }

    function partnerPresenceUpdate() {
      return resource.data.partnerUid == request.auth.uid
        && unchanged('partnerUid')
        && unchanged('hostName')
        && unchanged('partnerName')
        && unchanged('eveningTitle')
        && unchanged('version')
        && unchanged('status')
        && unchanged('lastEventId')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['partnerReady', 'partnerLastSeenMs', 'updatedAtMs', 'updatedBy', 'updatedAt']);
    }

    function partnerLeaveUpdate() {
      return resource.data.partnerUid == request.auth.uid
        && request.resource.data.partnerUid == null
        && request.resource.data.partnerName == null
        && request.resource.data.partnerReady == false
        && request.resource.data.partnerLastSeenMs == 0
        && request.resource.data.status == 'waiting'
        && unchanged('hostName')
        && unchanged('eveningTitle')
        && unchanged('version')
        && unchanged('lastEventId')
        && unchanged('hostReady')
        && unchanged('hostLastSeenMs')
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            'partnerUid',
            'partnerName',
            'partnerReady',
            'partnerLastSeenMs',
            'status',
            'updatedAtMs',
            'updatedBy',
            'updatedAt'
          ]);
    }

    function validUserProfile() {
      return request.resource.data.uid == request.auth.uid
        && request.resource.data.displayName is string
        && request.resource.data.displayName.size() > 0
        && request.resource.data.displayName.size() <= 32
        && request.resource.data.partnerDisplayName is string
        && request.resource.data.partnerDisplayName.size() <= 32
        && request.resource.data.lastSeenMs is int
        && request.resource.data.updatedAtMs is int
        && request.resource.data.schemaVersion is int;
    }

    function userCreateSafe() {
      let data = request.resource.data;
      return data.keys().hasOnly(['uid', 'displayName', 'partnerDisplayName', 'avatar', 'username',
          'createdAt', 'createdAtMs', 'lastSeenMs', 'updatedAt', 'updatedAtMs',
          'schemaVersion', 'lastRoomId', 'gamesPlayed', 'banned', 'approved', 'adminNote'])
        && data.get('banned', false) == false && data.approved == false && data.get('adminNote', '') == ''
        && data.get('gamesPlayed', 0) == 0 && data.get('createdAtMs', 0) is int
        && data.get('avatar', '') is string && data.get('avatar', '').size() <= 8
        && data.get('username', '') is string && data.get('username', '').size() <= 32
        && (data.get('lastRoomId', null) == null || (data.lastRoomId is string && data.lastRoomId.size() <= 128))
        && (!('createdAt' in data) || data.createdAt is timestamp)
        && (!('updatedAt' in data) || data.updatedAt is timestamp);
    }

    function userSelfFieldLocks() {
      return unchanged('banned')
        && unchanged('approved')
        && unchanged('adminNote');
    }

    function userSelfUpdateKeysOk() {
      return request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly([
          'displayName',
          'partnerDisplayName',
          'avatar',
          'lastSeenMs',
          'updatedAtMs',
          'updatedAt',
          'gamesPlayed',
          'lastRoomId',
          'username',
          'schemaVersion',
          'createdAt',
          'createdAtMs'
        ]);
    }

    // Public site settings (login page needs registration toggle before auth)
    match /config/{docId} {
      allow read: if true;
      allow create, update: if isAdmin()
        && request.resource.data.updatedBy == request.auth.uid;
      allow delete: if false;
    }

    // Legacy admins collection — locked forever
    match /admins/{uid} {
      allow read, write: if false;
    }

    // Content bank overrides (questions/tasks edited in admin)
    match /contentItems/{itemId} {
      allow read: if isActiveAccount();
      allow create, update: if isAdmin()
        && request.resource.data.updatedByUid == request.auth.uid;
      allow delete: if isAdmin();
    }

    // Administrator notes are private, including during migration from old profiles.
    match /adminUserNotes/{userId} {
      allow read: if isAdmin();
      allow create, update: if isAdmin() && request.resource.data.keys().hasOnly(['note'])
        && request.resource.data.note is string && request.resource.data.note.size() <= 200;
    }

    match /users/{userId} {
      allow list: if isAdmin();
      allow get: if isAdmin() || (isEmailSignedIn() && request.auth.uid == userId
        && (!exists(/databases/$(database)/documents/users/$(userId)) || resource.data.get('adminNote', '') == ''));
      allow create: if isEmailSignedIn() && (
        (
          request.auth.uid == userId
          && registrationOpen()
          && validUserProfile()
          && userCreateSafe()
        )
        || (
          isAdmin()
          && request.resource.data.uid == userId
          && request.resource.data.get('adminNote', '') == ''
          && request.resource.data.displayName is string
          && request.resource.data.displayName.size() > 0
          && request.resource.data.displayName.size() <= 32
        )
      );
      allow update: if isEmailSignedIn()
        && unchanged('uid')
        && unchanged('createdAtMs')
        && (
          (
            request.auth.uid == userId
            && isActiveAccount()
            && userSelfFieldLocks()
            && userSelfUpdateKeysOk()
            && (
              validUserProfile()
              || request.resource.data.diff(resource.data).affectedKeys()
                .hasOnly(['lastSeenMs', 'updatedAtMs', 'updatedAt', 'gamesPlayed', 'lastRoomId', 'username'])
            )
          )
          || (
            isAdmin()
            && (unchanged('adminNote') || request.resource.data.get('adminNote', '') == '')
            // Sole admin account cannot be banned / unapproved via rules.
            && !(userId == 'tpbKWXtXWFapFC7Fd80Wd4IMqxC2'
              && (
                ('banned' in request.resource.data && request.resource.data.banned == true)
                || ('approved' in request.resource.data && request.resource.data.approved == false)
              ))
            && request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly([
                'banned',
                'adminNote',
                'displayName',
                'approved',
                'username',
                'updatedAtMs',
                'updatedAt'
              ])
          )
        );
      allow delete: if false;
    }

    // Only these four documents are used by the profile and cloud sync clients.
    match /users/{userId}/{collectionId}/{documentId} {
      function isCloudDocument() {
        return (collectionId == 'meta' && documentId in ['profile', 'health'])
          || (collectionId == 'data' && documentId in ['bundle', 'activeGame']);
      }
      allow get: if isCloudDocument() && (isAdmin() || (isActiveAccount() && request.auth.uid == userId));
      allow write: if isCloudDocument() && isActiveAccount() && request.auth.uid == userId;
    }

    // Admission, code reservations, counters and event writes belong to the room API.
    match /users/{userId}/coupleSpinBackup/{name} {
      function ownBackup() {
        return isActiveAccount() && request.auth.uid == userId && name in ['bundle', 'activeGame'];
      }
      allow get, delete: if ownBackup();
      allow create, update: if ownBackup() && request.resource.data.keys().hasOnly(['payload', 'updatedAtMs', 'updatedAt'])
        && request.resource.data.payload is string && request.resource.data.payload.toUtf8().size() <= 700000
        && request.resource.data.updatedAtMs is int && request.resource.data.updatedAt == request.time;
    }

    match /rooms/{roomId} {
      allow get: if isActiveAccount() && roomResourceActive()
        && (resource.data.hostUid == request.auth.uid || resource.data.partnerUid == request.auth.uid);

      allow update: if isActiveAccount() && roomResourceActive()
        && (hostPresenceUpdate() || partnerPresenceUpdate() || partnerLeaveUpdate());

      match /gameEvents/{eventId} {
        allow get: if isActiveAccount() && roomActive(roomId) && isRoomMember(roomId);
        allow list: if isActiveAccount() && roomActive(roomId) && isRoomMember(roomId)
          && request.query.limit > 0 && request.query.limit <= 32
          && request.query.get('offset', 0) == 0;
      }
    }
  }
}
