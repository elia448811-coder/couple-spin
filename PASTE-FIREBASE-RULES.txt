rules_version = '2';
// Couple Spin account, backup and room permissions.
service cloud.firestore {
  match /databases/{database}/documents {
    function isAuth() {
      return request.auth != null;
    }

    // The administrator is provisioned with this UID by a trusted operator.
    function isAdmin() {
      return isAuth() && request.auth.uid == 'tpbKWXtXWFapFC7Fd80Wd4IMqxC2'; // gitleaks:allow public Firebase account identifier
    }

    // Real account (blocks anonymous Auth used only for bootstrap).
    function isEmailSignedIn() {
      return isAuth()
        && request.auth.token.email is string
        && request.auth.token.email.size() > 3
        && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    function callerUserDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    // Account access requires a positive approval recorded by an administrator.
    function isActiveAccount() {
      return isAdmin() || (
        isEmailSignedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && callerUserDoc().data.get('banned', false) == false
        && callerUserDoc().data.get('approved', false) == true
      );
    }

    function unchanged(field) {
      return !(field in request.resource.data)
        || request.resource.data[field] == resource.data[field];
    }

    function roomPath(roomId) {
      return /databases/$(database)/documents/rooms/$(roomId);
    }

    function currentSyncPath(roomId) {
      return /databases/$(database)/documents/rooms/$(roomId)/sync/current;
    }

    function liveRoomMember(room) {
      return room.expiresAtMs > request.time.toMillis()
        && (room.hostUid == request.auth.uid || room.partnerUid == request.auth.uid);
    }

    function presenceOnly(room, after, roleReady, roleSeen) {
      let changed = after.diff(room).affectedKeys();
      return changed.hasOnly([roleReady, roleSeen, 'updatedAtMs', 'updatedAt', 'updatedBy'])
        && after[roleReady] is bool
        && after[roleSeen] is int
        && after.updatedAtMs is int && after.updatedAt == request.time
        && after.updatedBy == request.auth.uid;
    }

    function nextHostEvent(roomId, room, after) {
      let state = getAfter(currentSyncPath(roomId)).data;
      return after.diff(room).affectedKeys().hasOnly(['version', 'updatedAtMs', 'updatedAt', 'updatedBy', 'status'])
        && after.updatedBy == request.auth.uid && after.updatedAtMs is int && after.updatedAt == request.time
        && after.version is int && after.version == room.version + 1 && after.version <= 2048
        && state.version == after.version && state.createdAt == request.time
        && (
          (state.type == 'GAME_STARTED' && room.hostReady && room.partnerReady && after.status == 'playing')
          || (state.type == 'GAME_ENDED' && after.status == 'ended')
          || (state.type == 'GAME_SNAPSHOT' && after.status == room.status)
        );
    }

    function validHostSync(roomId, state) {
      let room = get(roomPath(roomId)).data;
      let after = getAfter(roomPath(roomId)).data;
      return liveRoomMember(room) && room.hostUid == request.auth.uid && room.partnerUid != null
        && state.keys().hasOnly(['version','createdAt','type','payload','coupleSpinExpireAt'])
        && state.version is int && state.version == room.version + 1 && state.version <= 2048
        && state.version == after.version && state.createdAt == request.time
        && state.coupleSpinExpireAt == timestamp.value(room.expiresAtMs)
        && state.payload is string && state.payload.toUtf8().size() <= 120000
        && ((state.type == 'GAME_STARTED' && state.payload == '')
          || (state.type in ['GAME_SNAPSHOT', 'GAME_ENDED'] && state.payload.size() > 0));
    }

    function boundedProfileFields() {
      let d = request.resource.data;
      return d.get('username', '') is string && d.get('username', '').size() <= 32
        && d.get('avatar', '') is string && d.get('avatar', '').size() <= 8
        && (d.get('lastRoomId', null) == null || (d.lastRoomId is string && d.lastRoomId.size() <= 128))
        && d.get('gamesPlayed', 0) is int && d.get('gamesPlayed', 0) >= 0
        && d.get('createdAtMs', 0) is int && d.get('lastSeenMs', 0) is int
        && d.get('updatedAtMs', 0) is int && d.get('createdAt', request.time) is timestamp
        && d.get('updatedAt', request.time) is timestamp;
    }

    function validUserProfile() {
      return request.resource.data.uid == request.auth.uid
        && request.resource.data.displayName is string
        && request.resource.data.displayName.size() > 0
        && request.resource.data.displayName.size() <= 32
        && request.resource.data.partnerDisplayName is string
        && request.resource.data.partnerDisplayName.size() <= 32
        && request.resource.data.lastSeenMs is int
        && request.resource.data.updatedAtMs is int
        && request.resource.data.schemaVersion is int;
    }

    function userCreateSafe() {
      return request.resource.data.keys().hasOnly([
          'uid','username','displayName','partnerDisplayName','avatar','createdAtMs','createdAt',
          'lastSeenMs','updatedAtMs','updatedAt','schemaVersion','lastRoomId','gamesPlayed','approved','banned'
        ])
        && request.resource.data.get('banned', false) == false
        && request.resource.data.approved == false
        && (!exists(/databases/$(database)/documents/config/app)
          || get(/databases/$(database)/documents/config/app).data.get('registrationEnabled', true) == true);
    }

    function userSelfFieldLocks() {
      return unchanged('banned')
        && unchanged('approved')
        && unchanged('adminNote');
    }

    function userSelfUpdateKeysOk() {
      return request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly([
          'displayName',
          'partnerDisplayName',
          'avatar',
          'lastSeenMs',
          'updatedAtMs',
          'updatedAt',
          'gamesPlayed',
          'lastRoomId',
          'username',
          'schemaVersion',
          'createdAt',
          'createdAtMs'
        ]);
    }

    // Public site settings (login page needs registration toggle before auth)
    match /config/{docId} {
      allow read: if true;
      allow create, update: if isAdmin()
        && request.resource.data.updatedBy == request.auth.uid;
      allow delete: if false;
    }

    // Legacy admins collection — locked forever
    match /admins/{uid} {
      allow read, write: if false;
    }

    // Content bank overrides (questions/tasks edited in admin)
    match /contentItems/{itemId} {
      allow read: if isActiveAccount();
      allow create, update: if isAdmin()
        && request.resource.data.updatedByUid == request.auth.uid;
      allow delete: if isAdmin();
    }

    // Root user account document
    match /users/{userId} {
      allow read: if isAdmin();
      // Legacy profiles with a private note are admin-only until migrated.
      allow get: if isEmailSignedIn() && request.auth.uid == userId
        && (resource == null || resource.data.get('adminNote', '') == '');
      allow create: if isEmailSignedIn() && (
        (
          request.auth.uid == userId
          && validUserProfile()
          && userCreateSafe()
          && boundedProfileFields()
        )
        || (
          isAdmin()
          && !('adminNote' in request.resource.data)
          && request.resource.data.uid == userId
          && request.resource.data.displayName is string
          && request.resource.data.displayName.size() > 0
          && request.resource.data.displayName.size() <= 32
        )
      );
      allow update: if isEmailSignedIn()
        && unchanged('uid')
        && unchanged('createdAtMs')
        && (
          (
            request.auth.uid == userId
            && userSelfFieldLocks()
            && userSelfUpdateKeysOk()
            && boundedProfileFields()
            && (
              validUserProfile()
              || request.resource.data.diff(resource.data).affectedKeys()
                .hasOnly(['lastSeenMs', 'updatedAtMs', 'updatedAt', 'gamesPlayed', 'lastRoomId', 'username'])
            )
          )
          || (
            isAdmin()
            && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['adminNote'])
              || !('adminNote' in request.resource.data))
            // Sole admin account cannot be banned / unapproved via rules.
            && !(userId == 'tpbKWXtXWFapFC7Fd80Wd4IMqxC2'
              && (
                ('banned' in request.resource.data && request.resource.data.banned == true)
                || ('approved' in request.resource.data && request.resource.data.approved == false)
              ))
            && request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly([
                'banned',
                'adminNote',
                'displayName',
                'approved',
                'username',
                'updatedAtMs',
                'updatedAt'
              ])
          )
        );
      allow delete: if false;
    }

    match /adminUserNotes/{uid} {
      allow read: if isAdmin();
      allow create, update: if isAdmin() && request.resource.data.keys().hasOnly(['note'])
        && request.resource.data.note is string && request.resource.data.note.size() <= 200;
    }

    match /users/{userId}/meta/health {
      allow get: if isActiveAccount() && request.auth.uid == userId;
      allow create, update: if isActiveAccount() && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['ok','at','schemaVersion','client'])
        && request.resource.data.ok == true && request.resource.data.at == request.time
        && request.resource.data.schemaVersion is int && request.resource.data.client == 'couple-spin';
    }

    match /users/{userId}/data/{name} {
      allow get: if isActiveAccount() && request.auth.uid == userId && name in ['bundle','activeGame'];
      allow create, update: if isActiveAccount() && request.auth.uid == userId && name in ['bundle','activeGame']
        && request.resource.data.keys().hasOnly(['payload','updatedAtMs','updatedAt'])
        && request.resource.data.updatedAt == request.time && request.resource.data.updatedAtMs is int
        && request.resource.data.payload is string && request.resource.data.payload.toUtf8().size() <= 700000;
      allow delete: if isActiveAccount() && request.auth.uid == userId && name in ['bundle','activeGame'];
    }

    // Admission, reservations, and per-account budgets are managed by the room callable.
    match /rooms/{roomId} {
      allow get: if isActiveAccount() && liveRoomMember(resource.data);
      allow update: if isActiveAccount() && liveRoomMember(resource.data) && (
        (resource.data.hostUid == request.auth.uid && (
          presenceOnly(resource.data, request.resource.data, 'hostReady', 'hostLastSeenMs')
          || nextHostEvent(roomId, resource.data, request.resource.data)
        ))
        || (resource.data.partnerUid == request.auth.uid &&
          presenceOnly(resource.data, request.resource.data, 'partnerReady', 'partnerLastSeenMs'))
      );

      // A single replaceable snapshot bounds storage and never replays retired events.
      match /sync/current {
        allow get: if isActiveAccount() && liveRoomMember(get(roomPath(roomId)).data);
        allow create, update: if isActiveAccount() && validHostSync(roomId, request.resource.data);
      }
    }
  }
}
